cmake_minimum_required(VERSION 3.16)
project(my_capi_app LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(capi 0.4 CONFIG REQUIRED COMPONENTS sqlite templates)

# Only this small registry is generated during configuration. Controllers stay
# ordinary editable source files; the CLI never rewrites them or this file.
set(manifest "${CMAKE_CURRENT_SOURCE_DIR}/.capi/resources")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}")
file(STRINGS "${manifest}" resources)
set(registry "#include \"resources.h\"\n")
set(registrations "")
set(sources main.c web.c)
set(seen)
foreach(resource IN LISTS resources)
  string(LENGTH "${resource}" length)
  if(NOT resource MATCHES "^[a-z][a-z0-9_]*$" OR resource MATCHES "^sqlite_" OR length GREATER 32 OR resource IN_LIST seen)
    message(FATAL_ERROR "Invalid or duplicate resource in .capi/resources: ${resource}")
  endif()
  list(APPEND seen "${resource}")
  list(APPEND sources "resources/${resource}.c")
  string(APPEND registry "#include \"resources/${resource}.h\"\n")
  string(APPEND registrations "    if (resource_${resource}_register(app, db)) return -1;\n")
endforeach()
string(APPEND registry "int app_resources_register(capi_app *app, sqlite3 *db) {\n    (void)app; (void)db;\n${registrations}    return 0;\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/resources.c" "${registry}")
add_executable(app ${sources} "${CMAKE_CURRENT_BINARY_DIR}/resources.c")
target_include_directories(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(app PRIVATE capi::sqlite capi::templates)
target_compile_definitions(app PRIVATE TODO_VIEW_DIR="${CMAKE_CURRENT_SOURCE_DIR}/views")
