cmake_minimum_required(VERSION 3.16)
project(capi VERSION 0.2.0 LANGUAGES C)

option(CAPI_BUILD_EXAMPLES "Build example applications" ON)
option(CAPI_BUILD_TESTS "Build tests" ON)
option(CAPI_BUILD_TOOLS "Build scaffolder and load tool" ON)
option(CAPI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CAPI_FORCE_POLL "Use the portable poll backend" OFF)
option(CAPI_BUILD_FUZZER "Build libFuzzer HTTP harness (Clang only)" OFF)
option(CAPI_WITH_SQLITE "Build the optional SQLite adapter (requires SQLite3)" OFF)
option(CAPI_WITH_TEMPLATES "Build the optional HTML template engine" ON)

if(NOT UNIX)
  message(FATAL_ERROR "capi currently supports POSIX systems (Linux, macOS, BSD)")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CAPI_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes)
endif()

add_library(capi src/capi.c src/http.c src/json.c src/server.c)
add_library(capi::capi ALIAS capi)
target_compile_features(capi PUBLIC c_std_11)
target_include_directories(capi PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                      $<INSTALL_INTERFACE:include>)
if(CAPI_FORCE_POLL)
  target_compile_definitions(capi PRIVATE CAPI_FORCE_POLL=1)
endif()
set_target_properties(capi PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)

if(CAPI_WITH_SQLITE)
  find_package(SQLite3 3.8.7 REQUIRED)
  # Older CMake releases provide only the older imported target spelling.
  if(NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 INTERFACE IMPORTED)
    target_link_libraries(SQLite3::SQLite3 INTERFACE SQLite::SQLite3)
  endif()
  add_library(capi_sqlite extras/sqlite.c)
  add_library(capi::sqlite ALIAS capi_sqlite)
  target_link_libraries(capi_sqlite PUBLIC capi SQLite3::SQLite3)
  set_target_properties(capi_sqlite PROPERTIES EXPORT_NAME sqlite VERSION ${PROJECT_VERSION} SOVERSION 0)
endif()
if(CAPI_WITH_TEMPLATES)
  add_library(capi_templates extras/template.c)
  add_library(capi::templates ALIAS capi_templates)
  target_link_libraries(capi_templates PUBLIC capi)
  set_target_properties(capi_templates PROPERTIES EXPORT_NAME templates VERSION ${PROJECT_VERSION} SOVERSION 0)
endif()

if(CAPI_BUILD_EXAMPLES)
  add_executable(capi_hello examples/hello.c)
  target_link_libraries(capi_hello PRIVATE capi)
  add_executable(capi_api examples/api.c)
  target_link_libraries(capi_api PRIVATE capi)
  if(CAPI_WITH_SQLITE AND CAPI_WITH_TEMPLATES)
    add_executable(capi_notes examples/notes/main.c examples/notes/app.c)
    target_link_libraries(capi_notes PRIVATE capi_sqlite capi_templates)
    target_compile_definitions(capi_notes PRIVATE CAPI_NOTES_VIEW_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples/notes/views")
  endif()
endif()

if(CAPI_BUILD_TOOLS)
  add_executable(capi_cli tools/capi.c)
  set_target_properties(capi_cli PROPERTIES OUTPUT_NAME capi)
  add_executable(capi_bench tools/bench.c)
endif()

if(CAPI_BUILD_TESTS)
  enable_testing()
  add_executable(capi_tests tests/test.c)
  target_link_libraries(capi_tests PRIVATE capi)
  add_test(NAME core COMMAND capi_tests)
  add_executable(capi_network_tests tests/network.c)
  target_link_libraries(capi_network_tests PRIVATE capi)
  add_test(NAME network COMMAND capi_network_tests)
  set_tests_properties(core network PROPERTIES TIMEOUT 30)
  if(CAPI_WITH_TEMPLATES)
    add_executable(capi_template_tests tests/template.c)
    target_link_libraries(capi_template_tests PRIVATE capi_templates)
    add_test(NAME templates COMMAND capi_template_tests)
  endif()
  if(CAPI_WITH_SQLITE)
    add_executable(capi_sqlite_tests tests/sqlite.c)
    target_link_libraries(capi_sqlite_tests PRIVATE capi_sqlite)
    add_test(NAME sqlite COMMAND capi_sqlite_tests)
  endif()
  if(CAPI_WITH_SQLITE AND CAPI_WITH_TEMPLATES)
    add_executable(capi_notes_tests tests/notes.c examples/notes/app.c)
    target_link_libraries(capi_notes_tests PRIVATE capi_sqlite capi_templates)
    target_compile_definitions(capi_notes_tests PRIVATE CAPI_NOTES_VIEW_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples/notes/views")
    add_test(NAME notes COMMAND capi_notes_tests)
  endif()
endif()

if(CAPI_BUILD_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The fuzz target requires Clang with libFuzzer")
  endif()
  target_compile_options(capi PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(capi INTERFACE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(capi_fuzz tests/fuzz.c)
  target_compile_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(capi_fuzz PRIVATE capi)
  if(CAPI_WITH_TEMPLATES)
    target_compile_options(capi_templates PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
    add_executable(capi_template_fuzz tests/template_fuzz.c)
    target_compile_options(capi_template_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(capi_template_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(capi_template_fuzz PRIVATE capi_templates)
  endif()
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS capi EXPORT capiTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(CAPI_BUILD_TOOLS)
  install(TARGETS capi_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES include/capi.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(CAPI_WITH_TEMPLATES)
  install(TARGETS capi_templates EXPORT capiTemplateTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
          LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(EXPORT capiTemplateTargets NAMESPACE capi:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
  install(FILES include/capi_template.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  configure_file(cmake/capi-templates.pc.in capi-templates.pc @ONLY)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capi-templates.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
endif()
if(CAPI_WITH_SQLITE)
  install(TARGETS capi_sqlite EXPORT capiSqliteTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
          LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(EXPORT capiSqliteTargets NAMESPACE capi:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
  install(FILES include/capi_sqlite.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  configure_file(cmake/capi-sqlite.pc.in capi-sqlite.pc @ONLY)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capi-sqlite.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
endif()
install(EXPORT capiTargets NAMESPACE capi:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
configure_package_config_file(cmake/capiConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/capiConfig.cmake
                             INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/capiConfigVersion.cmake
                                COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capiConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/capiConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
configure_file(cmake/capi.pc.in capi.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capi.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
