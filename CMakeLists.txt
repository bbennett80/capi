cmake_minimum_required(VERSION 3.16)
project(capi VERSION 0.1.0 LANGUAGES C)

option(CAPI_BUILD_EXAMPLES "Build example applications" ON)
option(CAPI_BUILD_TESTS "Build tests" ON)
option(CAPI_BUILD_TOOLS "Build scaffolder and load tool" ON)
option(CAPI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CAPI_FORCE_POLL "Use the portable poll backend" OFF)
option(CAPI_BUILD_FUZZER "Build libFuzzer HTTP harness (Clang only)" OFF)

if(NOT UNIX)
  message(FATAL_ERROR "capi currently supports POSIX systems (Linux, macOS, BSD)")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CAPI_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes)
endif()

add_library(capi src/capi.c src/http.c src/json.c src/server.c)
add_library(capi::capi ALIAS capi)
target_compile_features(capi PUBLIC c_std_11)
target_include_directories(capi PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                      $<INSTALL_INTERFACE:include>)
if(CAPI_FORCE_POLL)
  target_compile_definitions(capi PRIVATE CAPI_FORCE_POLL=1)
endif()
set_target_properties(capi PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)

if(CAPI_BUILD_EXAMPLES)
  add_executable(capi_hello examples/hello.c)
  target_link_libraries(capi_hello PRIVATE capi)
  add_executable(capi_api examples/api.c)
  target_link_libraries(capi_api PRIVATE capi)
endif()

if(CAPI_BUILD_TOOLS)
  add_executable(capi_cli tools/capi.c)
  set_target_properties(capi_cli PROPERTIES OUTPUT_NAME capi)
  add_executable(capi_bench tools/bench.c)
endif()

if(CAPI_BUILD_TESTS)
  enable_testing()
  add_executable(capi_tests tests/test.c)
  target_link_libraries(capi_tests PRIVATE capi)
  add_test(NAME core COMMAND capi_tests)
  add_executable(capi_network_tests tests/network.c)
  target_link_libraries(capi_network_tests PRIVATE capi)
  add_test(NAME network COMMAND capi_network_tests)
  set_tests_properties(core network PROPERTIES TIMEOUT 30)
endif()

if(CAPI_BUILD_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The fuzz target requires Clang with libFuzzer")
  endif()
  target_compile_options(capi PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(capi INTERFACE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(capi_fuzz tests/fuzz.c)
  target_compile_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(capi_fuzz PRIVATE capi)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS capi EXPORT capiTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(CAPI_BUILD_TOOLS)
  install(TARGETS capi_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES include/capi.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT capiTargets NAMESPACE capi:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
configure_package_config_file(cmake/capiConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/capiConfig.cmake
                             INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/capiConfigVersion.cmake
                                COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capiConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/capiConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/capi)
configure_file(cmake/capi.pc.in capi.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/capi.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
