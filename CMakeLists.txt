cmake_minimum_required(VERSION 3.16)
project(capi VERSION 0.4.0 LANGUAGES C)
# Pre-1.0 minor releases can change C struct layouts; keep their shared ABIs distinct.
set(CAPI_ABI_VERSION "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}")

option(CAPI_BUILD_EXAMPLES "Build example applications" ON)
option(CAPI_BUILD_TESTS "Build tests" ON)
option(CAPI_BUILD_TOOLS "Build scaffolder and load tool" ON)
option(CAPI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CAPI_FORCE_POLL "Use the portable poll backend" OFF)
option(CAPI_BUILD_FUZZER "Build libFuzzer HTTP harness (Clang only)" OFF)
option(CAPI_WITH_SQLITE "Build the optional SQLite adapter (requires SQLite3)" OFF)
option(CAPI_WITH_TEMPLATES "Build the optional HTML template engine" ON)
option(CAPI_WITH_WORKERS "Build the optional POSIX worker pool" OFF)
option(CAPI_THREAD_SANITIZE "Enable ThreadSanitizer (separate from ASan/fuzzing)" OFF)

if(NOT UNIX)
  message(FATAL_ERROR "capi currently supports POSIX systems (Linux, macOS, BSD)")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CAPI_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CAPI_THREAD_SANITIZE)
  if(CAPI_SANITIZE OR CAPI_BUILD_FUZZER)
    message(FATAL_ERROR "ThreadSanitizer must use a separate build from ASan/libFuzzer")
  endif()
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes)
endif()

set(CAPI_CORE_SOURCES src/capi.c src/http.c src/input.c src/json.c src/observe.c src/router.c src/server.c)
add_library(capi ${CAPI_CORE_SOURCES})
add_library(capi::capi ALIAS capi)
target_compile_features(capi PUBLIC c_std_11)
target_include_directories(capi PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                      $<INSTALL_INTERFACE:include>)
if(CAPI_FORCE_POLL)
  target_compile_definitions(capi PRIVATE CAPI_FORCE_POLL=1)
endif()
set_target_properties(capi PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${CAPI_ABI_VERSION})

if(CAPI_WITH_SQLITE)
  find_package(SQLite3 3.8.7 REQUIRED)
  # Older CMake releases provide only the older imported target spelling.
  if(NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 INTERFACE IMPORTED)
    target_link_libraries(SQLite3::SQLite3 INTERFACE SQLite::SQLite3)
  endif()
  add_library(capi_sqlite extras/sqlite.c)
  add_library(capi::sqlite ALIAS capi_sqlite)
  target_link_libraries(capi_sqlite PUBLIC capi SQLite3::SQLite3)
  set_target_properties(capi_sqlite PROPERTIES EXPORT_NAME sqlite VERSION ${PROJECT_VERSION} SOVERSION ${CAPI_ABI_VERSION})
endif()
if(CAPI_WITH_TEMPLATES)
  add_library(capi_templates extras/template.c)
  add_library(capi::templates ALIAS capi_templates)
  target_link_libraries(capi_templates PUBLIC capi)
  set_target_properties(capi_templates PROPERTIES EXPORT_NAME templates VERSION ${PROJECT_VERSION} SOVERSION ${CAPI_ABI_VERSION})
endif()
if(CAPI_WITH_WORKERS)
  set(THREADS_PREFER_PTHREAD_FLAG TRUE)
  find_package(Threads REQUIRED)
  add_library(capi_workers extras/workers.c)
  add_library(capi::workers ALIAS capi_workers)
  target_link_libraries(capi_workers PUBLIC capi Threads::Threads)
  set_target_properties(capi_workers PROPERTIES EXPORT_NAME workers VERSION ${PROJECT_VERSION} SOVERSION ${CAPI_ABI_VERSION})
endif()

if(CAPI_BUILD_TOOLS)
  # Embed line arrays so the installed C CLI needs no template directory.
  set(scaffold_header "${CMAKE_CURRENT_BINARY_DIR}/generated/scaffold_templates.h")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  file(WRITE "${scaffold_header}" "/* Generated from tools/templates; edit the source templates. */\n")
  function(capi_embed_template name path)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
    file(READ "${path}" contents)
    string(REPLACE "\\" "\\\\" contents "${contents}")
    string(REPLACE "\"" "\\\"" contents "${contents}")
    string(REPLACE "\n" "\\n\",\n\"" contents "${contents}")
    file(APPEND "${scaffold_header}" "static const char *const ${name}[] = {\n\"${contents}\",\nNULL};\n")
  endfunction()
  capi_embed_template(sqlite_main tools/templates/sqlite_main.c)
  capi_embed_template(sqlite_cmake tools/templates/sqlite_CMakeLists.txt)
  capi_embed_template(sqlite_readme tools/templates/sqlite_README.md)
  capi_embed_template(resource_template tools/templates/resource.c)
  add_executable(capi_cli tools/capi.c tools/generate.c)
  target_include_directories(capi_cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  set_target_properties(capi_cli PROPERTIES OUTPUT_NAME capi)
  add_executable(capi_bench tools/bench.c)
  add_executable(capi_perf_report tools/perf_report.c)
  if(CAPI_WITH_SQLITE AND CAPI_WITH_TEMPLATES)
    add_executable(capi_perf_server tools/perf_server.c)
    target_link_libraries(capi_perf_server PRIVATE capi_sqlite capi_templates)
  endif()
endif()

  if(CAPI_BUILD_TOOLS)
    find_package(Git QUIET)
    if(GIT_FOUND)

  if(CAPI_BUILD_TOOLS AND CAPI_WITH_SQLITE)
    add_test(NAME cli COMMAND ${CMAKE_COMMAND}
      "-DCLI=$<TARGET_FILE:capi_cli>" "-DBINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}"
      "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}" "-DCC=${CMAKE_C_COMPILER}"
      "-DCONFIG=$<CONFIG>" "-DCOMPILE_FLAGS=${CMAKE_C_FLAGS}"
      "-DSANITIZE=${CAPI_SANITIZE}" "-DTHREAD_SANITIZE=${CAPI_THREAD_SANITIZE}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
    set_tests_properties(cli PROPERTIES TIMEOUT 120)
  endif()
  add_executable(capi_tests tests/test.c)
  target_link_libraries(capi_tests PRIVATE capi)
  add_test(NAME core COMMAND capi_tests)
  add_executable(capi_router_tests tests/router.c)
  target_link_libraries(capi_router_tests PRIVATE capi)
  add_test(NAME routers COMMAND capi_router_tests)
  add_executable(capi_reliability_tests tests/reliability.c)
  target_link_libraries(capi_reliability_tests PRIVATE capi)
  add_test(NAME reliability COMMAND capi_reliability_tests)
  add_library(capi_fault_core STATIC ${CAPI_CORE_SOURCES})
  target_include_directories(capi_fault_core PUBLIC include)
  target_compile_definitions(capi_fault_core PRIVATE malloc=cp_test_malloc calloc=cp_test_calloc realloc=cp_test_realloc)

    target_compile_definitions(capi_fault_core PRIVATE CAPI_FORCE_POLL=1)

  add_executable(capi_fault_tests tests/faults.c)

  add_test(NAME allocation_faults COMMAND capi_fault_tests)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_test(NAME compiler_contracts COMMAND ${CMAKE_COMMAND}
      -DCC=${CMAKE_C_COMPILER} -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
      -DWORKERS=${CAPI_WITH_WORKERS} -DSQLITE=${CAPI_WITH_SQLITE}
      "-DSQLITE_INCLUDES=${SQLite3_INCLUDE_DIRS}" -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_checks.cmake)
  endif()
  add_executable(capi_network_tests tests/network.c)
  target_link_libraries(capi_network_tests PRIVATE capi)
  add_test(NAME network COMMAND capi_network_tests)
  set_tests_properties(core network PROPERTIES TIMEOUT 30)
  if(CAPI_WITH_WORKERS)
    add_executable(capi_worker_tests tests/workers.c)
    target_link_libraries(capi_worker_tests PRIVATE capi_workers)
    add_test(NAME workers COMMAND capi_worker_tests)
    set_tests_properties(workers PROPERTIES TIMEOUT 30)
    if(CAPI_WITH_SQLITE AND CAPI_BUILD_EXAMPLES)
      add_executable(capi_async_sqlite_tests tests/async_sqlite.c)
      add_test(NAME async_sqlite COMMAND capi_async_sqlite_tests $<TARGET_FILE:capi_async_sqlite>)
      set_tests_properties(async_sqlite PROPERTIES TIMEOUT 30)
    endif()
  endif()
  if(CAPI_WITH_TEMPLATES)
    add_executable(capi_template_tests tests/template.c)
    target_link_libraries(capi_template_tests PRIVATE capi_templates)
    add_test(NAME templates COMMAND capi_template_tests)

  if(CAPI_WITH_SQLITE)
    add_executable(capi_sqlite_tests tests/sqlite.c)
    target_link_libraries(capi_sqlite_tests PRIVATE capi_sqlite)
    add_test(NAME sqlite COMMAND capi_sqlite_tests)
  endif()
  if(CAPI_WITH_SQLITE AND CAPI_WITH_TEMPLATES)

    target_include_directories(capi_todo_tests PRIVATE examples/todo)
    target_link_libraries(capi_todo_tests PRIVATE capi_sqlite capi_templates)
    target_compile_definitions(capi_todo_tests PRIVATE TODO_VIEW_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples/todo/views")
    add_test(NAME todo_web COMMAND capi_todo_tests)
    add_executable(capi_notes_tests tests/notes.c examples/notes/app.c)
    target_link_libraries(capi_notes_tests PRIVATE capi_sqlite capi_templates)
    target_compile_definitions(capi_notes_tests PRIVATE CAPI_NOTES_VIEW_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples/notes/views")

if(CAPI_BUILD_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The fuzz target requires Clang with libFuzzer")
  endif()
  target_compile_options(capi PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(capi INTERFACE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(capi_fuzz tests/fuzz.c)
  target_compile_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(capi_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(capi_fuzz PRIVATE capi)
  if(CAPI_WITH_TEMPLATES)
    target_compile_options(capi_templates PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
    add_executable(capi_template_fuzz tests/template_fuzz.c)
    target_compile_options(capi_template_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(capi_template_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(capi_template_fuzz PRIVATE capi_templates)
  endif()
endif()
