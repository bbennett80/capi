cmake_minimum_required(VERSION 3.16)
project(capi_consumer LANGUAGES C)
option(CAPI_CONSUME_EXTRAS "Consume installed SQLite and template components" OFF)
option(CAPI_CONSUME_WORKERS "Consume installed worker component" OFF)
set(components)
if(CAPI_CONSUME_EXTRAS)
  list(APPEND components sqlite templates)
endif()
if(CAPI_CONSUME_WORKERS)
  list(APPEND components workers)
endif()
if(components)
  find_package(capi 0.3 CONFIG REQUIRED COMPONENTS ${components})
else()
  find_package(capi 0.3 CONFIG REQUIRED)
  if(TARGET capi::sqlite OR TARGET capi::templates OR TARGET capi::workers)
    message(FATAL_ERROR "Optional components leaked into a core-only consumer")
  endif()
endif()
add_executable(consumer main.c)
if(CAPI_CONSUME_EXTRAS)
  target_compile_definitions(consumer PRIVATE CONSUME_EXTRAS=1)
  target_link_libraries(consumer PRIVATE capi::sqlite capi::templates)
elseif(NOT CAPI_CONSUME_WORKERS)
  target_link_libraries(consumer PRIVATE capi::capi)
endif()
if(CAPI_CONSUME_WORKERS)
  target_compile_definitions(consumer PRIVATE CONSUME_WORKERS=1)
  target_link_libraries(consumer PRIVATE capi::workers)
endif()
