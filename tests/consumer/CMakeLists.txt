cmake_minimum_required(VERSION 3.16)
project(capi_consumer LANGUAGES C)
option(CAPI_CONSUME_EXTRAS "Consume installed SQLite and template components" OFF)
if(CAPI_CONSUME_EXTRAS)
  find_package(capi 0.2 CONFIG REQUIRED COMPONENTS sqlite templates)
else()
  find_package(capi 0.2 CONFIG REQUIRED)
  if(TARGET capi::sqlite OR TARGET capi::templates)
    message(FATAL_ERROR "Optional components leaked into a core-only consumer")
  endif()
endif()
add_executable(consumer main.c)
if(CAPI_CONSUME_EXTRAS)
  target_compile_definitions(consumer PRIVATE CONSUME_EXTRAS=1)
  target_link_libraries(consumer PRIVATE capi::sqlite capi::templates)
else()
  target_link_libraries(consumer PRIVATE capi::capi)
endif()
